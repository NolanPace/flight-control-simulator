cmake_minimum_required(VERSION 3.16)
project(FlightControl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(flight STATIC flight.cpp)
target_include_directories(flight PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(flight_sim main.cpp)
target_link_libraries(flight_sim PRIVATE flight)
enable_testing()
add_executable(flight_tests tests.cpp)
target_link_libraries(flight_tests PRIVATE flight)
add_test(NAME flight_tests COMMAND flight_tests)
